use std::sync::mpsc; // SyncSender and Receiver live here
use std::time::{Duration, Instant};
use std::io::{BufRead, BufReader}; // Removed 'self' to fix the warning
use std::thread;
use serde::Deserialize;

#[derive(Deserialize, Debug)] // Zero-copy structs using lifetimes <'a>
pub struct WikipediaEdit<'a> {
    #[serde(rename = "server_name")]
    pub server: &'a str, 
    pub user: &'a str,
    pub bot: bool,
}

// FIX: Change 'mpsc::Sender' to 'mpsc::SyncSender' to match sync_channel
fn start_ingestion(tx: mpsc::SyncSender<String>) {
    let url = "https://stream.wikimedia.org/v2/stream/recentchange";
    
    let client = reqwest::blocking::Client::builder()
        .user_agent("RTS-Assignment-Student-Monitor/1.0 (Contact: student@apu.edu.my)")
        .build()
        .expect("Failed to build HTTP client");

    loop {
        println!("Connecting to Wikipedia (Threaded)...");
        let res = client.get(url).send();

        match res {
            Ok(response) => {
                println!("Connected! Monitoring firehose...");
                let reader = BufReader::new(response);

                for line in reader.lines() {
                    if let Ok(l) = line {
                        if l.starts_with("data: ") {
                            let json_str = l[6..].to_string();
                            
                            // Requirement: Bounded Behavior (Drop oldest/newest on overflow)
                            // try_send fails immediately if the 100-slot buffer is full
                            if let Err(_) = tx.try_send(json_str) {
                                eprintln!("[{:?}] OVERFLOW: Dropping packet", Instant::now());
                            }
                        }
                    }
                }
            }
            Err(e) => {
                eprintln!("Connection failed: {}. Retrying...", e);
            }
        }
        
        eprintln!("Stream interrupted. Retrying in 2 seconds...");
        thread::sleep(Duration::from_secs(2));
    }
}

fn main() {
    // Creates a bounded channel with 100 slots
    let (tx, rx) = mpsc::sync_channel::<String>(100);

    thread::spawn(move || {
        start_ingestion(tx);
    });

    println!("System Active (Threaded). 10s Watchdog engaged.");

    loop {
        // Requirement: Watchdog Timer via recv_timeout
        match rx.recv_timeout(Duration::from_secs(10)) {
            Ok(raw_json) => {
                let start = Instant::now();
                
                // Requirement: Zero-Copy Parsing
                if let Ok(edit) = serde_json::from_str::<WikipediaEdit>(&raw_json) {
                    let duration = start.elapsed();
                    
                    // Requirement: 2ms Deadline
                    if duration > Duration::from_millis(2) {
                        println!("[VIOLATION] {} took {:?}", edit.server, duration);
                    } else {
                        println!("[OK] {} | User: {} | Latency: {:?}", edit.server, edit.user, duration);
                    }
                }
            }
            Err(mpsc::RecvTimeoutError::Timeout) => {
                println!("[{:?}] WATCHDOG: No data for 10s.", Instant::now());
            }
            Err(mpsc::RecvTimeoutError::Disconnected) => {
                println!("Channel closed. Exiting...");
                break;
            }
        }
    }
}
